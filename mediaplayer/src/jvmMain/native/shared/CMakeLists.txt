cmake_minimum_required(VERSION 3.21)
project(NativeMediaRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(JNI REQUIRED)
if(NOT DEFINED GSTREAMER_ROOT)
  message(FATAL_ERROR "GSTREAMER_ROOT must point to the pinned 1.28.8 SDK")
endif()
add_library(NativeMediaRuntime SHARED NativeMediaRuntime.cpp WidevineHost.cpp)
target_include_directories(NativeMediaRuntime PRIVATE ${JNI_INCLUDE_DIRS} "${GSTREAMER_ROOT}/include" "${GSTREAMER_ROOT}/include/gstreamer-1.0" "${GSTREAMER_ROOT}/include/glib-2.0" "${GSTREAMER_ROOT}/lib/glib-2.0/include")
target_link_directories(NativeMediaRuntime PRIVATE "${GSTREAMER_ROOT}/lib")
target_link_libraries(NativeMediaRuntime PRIVATE gstreamer-1.0 gstapp-1.0 gstvideo-1.0 gstaudio-1.0 gstbase-1.0 gobject-2.0 glib-2.0 soup-3.0 ass)
target_compile_definitions(NativeMediaRuntime PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(NativeMediaBootstrap SHARED WindowsBootstrap.cpp)
target_include_directories(NativeMediaBootstrap PRIVATE ${JNI_INCLUDE_DIRS})
